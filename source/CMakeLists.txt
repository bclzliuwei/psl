cmake_minimum_required(VERSION 3.21)
project(BlockScheme VERSION 1.0.0 LANGUAGES CXX)

option(PSL_BUILD_GUI "Build the native Windows editor" ON)
include(CTest)

add_library(scheme_model scheme_model.cpp)
target_include_directories(scheme_model PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(scheme_model PUBLIC cxx_std_17)
if(MSVC)
    target_compile_options(scheme_model PRIVATE /W4 /permissive-)
endif()

if(PSL_BUILD_GUI)
    if(NOT WIN32)
        message(FATAL_ERROR "The editor requires Windows. Use -DPSL_BUILD_GUI=OFF to build model tests elsewhere.")
    endif()
    add_executable(BlockScheme WIN32 main.cpp)
    target_link_libraries(BlockScheme PRIVATE scheme_model gdiplus comdlg32 user32 gdi32)
    target_compile_definitions(BlockScheme PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    if(MSVC)
        target_compile_options(BlockScheme PRIVATE /W4 /permissive-)
    endif()
    install(TARGETS BlockScheme RUNTIME DESTINATION bin)
    if(BUILD_TESTING)
        add_test(NAME editor_smoke COMMAND BlockScheme --smoke-test)
        set_tests_properties(editor_smoke PROPERTIES TIMEOUT 30)
    endif()
endif()

if(BUILD_TESTING)
    add_executable(scheme_model_tests scheme_model_tests.cpp)
    target_link_libraries(scheme_model_tests PRIVATE scheme_model)
    add_test(NAME scheme_model COMMAND scheme_model_tests)
endif()
